cmake_minimum_required(VERSION 3.14)
project(allsense VERSION 0.1.0 LANGUAGES CXX)

# Use it from your project:
#   include(FetchContent)
#   FetchContent_Declare(allsense GIT_REPOSITORY https://github.com/wen-da-ng/otterbridge-allsense-cpp.git GIT_TAG main)
#   FetchContent_MakeAvailable(allsense)
#   target_link_libraries(my_robot PRIVATE allsense::allsense)
# or add_subdirectory(path/to/otterbridge-allsense-cpp).

set(top_level OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(top_level ON)
endif()
option(ALLSENSE_BUILD_EXAMPLES "Build the examples" ${top_level})
option(ALLSENSE_BUILD_TESTS "Build the conformance and unit test programs" ${top_level})

# JSON: an installed nlohmann_json if there is one, else download it (header-only, MIT).
# 3.11 or newer: older versions destroy deeply nested values recursively.
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
                       URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
  FetchContent_MakeAvailable(json)
endif()
find_package(Threads REQUIRED)

add_library(allsense src/allsense.cpp src/net.cpp)
add_library(allsense::allsense ALIAS allsense)
target_compile_features(allsense PUBLIC cxx_std_17)
target_include_directories(allsense PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(allsense PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
  target_link_libraries(allsense PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(allsense PRIVATE /W4 /utf-8)
else()
  target_compile_options(allsense PRIVATE -Wall -Wextra)
endif()

if(ALLSENSE_BUILD_EXAMPLES)
  foreach(example quickstart print_values log_csv)
    add_executable(${example} examples/${example}.cpp)
    target_link_libraries(${example} PRIVATE allsense::allsense)
    if(MSVC)
      target_compile_options(${example} PRIVATE /utf-8)
    endif()
  endforeach()
endif()

if(ALLSENSE_BUILD_TESTS)
  add_executable(conformance tests/conformance.cpp)
  target_link_libraries(conformance PRIVATE allsense::allsense)
  # Unit tests of the internals (limits, handshake, link parsing) against a loopback server: ctest or build/unit_tests.
  add_executable(unit_tests tests/unit.cpp)
  target_include_directories(unit_tests PRIVATE src)
  target_link_libraries(unit_tests PRIVATE allsense::allsense)
  if(WIN32)
    target_link_libraries(unit_tests PRIVATE ws2_32)
  endif()
  if(MSVC)
    target_compile_options(conformance PRIVATE /utf-8)
    target_compile_options(unit_tests PRIVATE /utf-8)
  endif()
  enable_testing()
  add_test(NAME unit_tests COMMAND unit_tests)
endif()
